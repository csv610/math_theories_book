\paragraph{The idea and history}
Topology grew from Euler's solution of the Seven Bridges of Königsberg problem and his polyhedron formula. Listing introduced the term topology; Cantor studied point sets; Fréchet, Hausdorff, and Kuratowski developed metric and general topological spaces. Dennis Sullivan's work illustrates the continuing interaction of algebraic, geometric, and dynamical topology \cite{topology_ref1}.

\section*{3. Theory}
\subsection*{Core theory}
\paragraph{Spaces and continuity}
A topology on a set $X$ is a collection of open sets containing $\varnothing$ and $X$, closed under arbitrary unions and finite intersections. A topological space is a set with such a collection. A metric defines a topology through open balls, but many useful topologies have no preferred distance.

A function is continuous when inverse images of open sets are open. A homeomorphism is a continuous bijection with continuous inverse; it means two spaces have the same topological shape. A homotopy continuously deforms one map into another. Compactness says that every open cover has a finite subcover, and connectedness means the space cannot be split into two disjoint open pieces \cite{topology_ref2}.

\paragraph{Examples and invariants}
\bookfigure{topology_deformation}{Topology studies features that survive continuous deformation.}
A circle and a square are homeomorphic, while a circle and two disjoint circles are not connected in the same way. A coffee cup and a torus have one hole each in the relevant sense. The Mobius strip has one side and one boundary component. Euler characteristic, fundamental group, homology, and cohomology turn shape into computable algebra.

General topology studies open sets and convergence. Algebraic topology studies invariants such as homology and homotopy groups. Differential topology studies smooth manifolds, transversality, and tangent spaces. Geometric topology studies knots, manifolds, and embeddings \cite{topology_ref3}.

\section*{4. Important theorems and results}
\begin{enumerate}[leftmargin=*,itemsep=0.35em]
\item \textbf{Euler characteristic.} For a finite cell decomposition, $\chi=V-E+F$ is independent of the chosen decomposition. \emph{Example.} A cube has $V=8$, $E=12$, $F=6$, giving $\chi=8-12+6=2$, matching the sphere.

\item \textbf{Intermediate value theorem.} A continuous real function taking values on both sides of a number takes that value somewhere between. \emph{Example.} If $f(0)=-1$ and $f(2)=3$, then $f(c)=0$ for some $c\in(0,2)$.

\item \textbf{Heine--Borel theorem.} A subset of Euclidean space is compact exactly when it is closed and bounded. \emph{Example.} $[0,1]$ is compact; $(0,1)$ is not (it is not closed); $[0,\infty)$ is not (it is not bounded).

\item \textbf{Brouwer fixed-point theorem.} Every continuous map from a closed disk to itself has a fixed point. \emph{Example.} Stirring coffee in a circular cup and waiting: some point ends up exactly where it started.

\item \textbf{Jordan curve theorem.} A simple closed curve in the plane separates the plane into an inside and an outside.

\item \textbf{Classification of surfaces.} Connected compact surfaces are classified by orientability, genus, and boundary data.
\end{enumerate}
\noindent\emph{Source for these results:} \cite{topology_ref1}.

\section*{Glossary}
\begin{description}[leftmargin=*,style=nextline,itemsep=0.3em]
\item[\textbf{Topological space.}] A set together with a collection of open sets that satisfy axioms for unions and intersections, providing a framework for continuity.
\item[\textbf{Open set.}] A member of the topology on a space; it defines the notion of continuity and convergence without requiring a distance.
\item[\textbf{Homeomorphism.}] A continuous bijection with a continuous inverse; it identifies two spaces that have the same topological shape.
\item[\textbf{Compactness.}] A property meaning every open cover of the space has a finite subcover; it generalizes closed and bounded subsets of Euclidean space.
\item[\textbf{Connectedness.}] A property meaning the space cannot be written as the union of two disjoint nonempty open sets.
\item[\textbf{Euler characteristic.}] For a finite cell decomposition, the number $\chi = V - E + F$, which is independent of the decomposition and is a topological invariant.
\item[\textbf{Fundamental group.}] The group of loops based at a point up to continuous deformation; it detects holes in a space.
\item[\textbf{Homotopy.}] A continuous deformation of one map into another; two maps are homotopic if one can be continuously transformed into the other.
\item[\textbf{Manifold.}] A space that locally resembles Euclidean space of some dimension.
\item[\textbf{Homology.}] An algebraic invariant assigning groups measuring the number of independent holes in each dimension.
\item[\textbf{Cohomology.}] An algebraic invariant dual to homology, supporting richer algebraic structures.
\item[\textbf{Jordan curve theorem.}] A simple closed curve in the plane separates the plane into interior and exterior.
\end{description}
\section*{7. Sample Questions}
\emph{Exercise reference:} \cite{topology_ref1}. The cited edition is the source for the exercise set; consult its Exercises section for the official statement and numbering.
\begin{enumerate}[leftmargin=*,itemsep=0.9em]
\item \textbf{Exercise 1.} Compute the Euler characteristic of the torus $T^2$ using a cell decomposition with one 0-cell, two 1-cells, and one 2-cell. \emph{Solution.}$V = 1$, $E = 2$, $F = 1$. Then $\chi(T^2) = V - E + F = 1 - 2 + 1 = 0$.

\item \textbf{Exercise 2.} Determine whether $\mathbb{R}^2$ and $\mathbb{R}^3$ are homeomorphic. \emph{Solution.} They are not. Remove a point from $\mathbb{R}^2$: the result $\mathbb{R}^2\setminus\{0\}$ has fundamental group $\pi_1=\mathbb{Z}$ (a loop around the puncture cannot be contracted). Remove a point from $\mathbb{R}^3$: the result $\mathbb{R}^3\setminus\{0\}$ is simply connected ($\pi_1=0$), since any loop can slip around the missing point in the extra dimension. Since homeomorphisms preserve fundamental groups, $\mathbb{R}^2\not\cong\mathbb{R}^3$.

\item \textbf{Exercise 3.} Show that $[0,1)$ and $(0,1)$ are not homeomorphic. \emph{Solution.} Remove the endpoint $0$ from $[0,1)$: the remainder $(0,1)$ is connected. Now remove any interior point $p\in(0,1)$ from $(0,1)$: it splits into two pieces $(0,p)\cup(p,1)$, so $(0,1)\setminus\{p\}$ is disconnected. Thus no point of $[0,1)$ is a ``disconnecting point,'' but every point of $(0,1)$ is. Since homeomorphisms preserve which points disconnect the space, $[0,1)\not\cong(0,1)$.

\item \textbf{Exercise 4.} Find the fundamental group $\pi_1(S^1 \times S^1)$ of the torus. \emph{Solution.} By the theorem for products, $\pi_1(X \times Y) \cong \pi_1(X) \times \pi_1(Y)$. Since $\pi_1(S^1) \cong \mathbb{Z}$, we get $\pi_1(S^1 \times S^1) \cong \mathbb{Z} \times \mathbb{Z}$. The two generators are the two standard loops around the torus, corresponding to the two independent directions.

\item \textbf{Exercise 5.} Prove that a continuous bijection from a compact space to a Hausdorff space is a homeomorphism. \emph{Solution.} Let $f\colon X \to Y$ be a continuous bijection with $X$ compact and $Y$ Hausdorff. We must show $f^{-1}$ is continuous, equivalently that $f$ is a closed map. Let $C\subseteq X$ be closed. Since $X$ is compact, $C$ is compact. Since $f$ is continuous, $f(C)$ is compact in $Y$. Since $Y$ is Hausdorff, every compact subset is closed, so $f(C)$ is closed. Thus $f$ sends closed sets to closed sets, proving $f^{-1}$ is continuous.

\end{enumerate}
